## Independent re-derivation commands used (all read-only, `TZ=America/Los_Angeles`)

```
# Fast-queue gaps (Figure 1, row 1) — reproduces MINING_REPORT.md §2.5
python3 - <<'PY'
lines = open('lab/rulings.txt').read().strip().split('\n')
ts = sorted(int(l.split('|')[0]) for l in lines)
bursts, cur = [], [ts[0]]
for t in ts[1:]:
    if t - cur[-1] <= 1800:
        cur.append(t)
    else:
        bursts.append(cur); cur = [t]
bursts.append(cur)
gaps = [round((b[i]-b[i-1])/60, 1) for b in bursts if len(b) >= 3 for i in range(1, len(b))]
print(len(gaps), sorted(gaps))
PY

# Present-to-ratify / confirmed-decided commit timestamps (Figure 1, rows 2-3)
git -C /workspace/whomp show -s --pretty='%h %ad %s' --date=format:'%m-%d %H:%M:%S' \
  fbab99ef e32a01e2 5ebf68c1 2e20b31d a9eb3369 86eb219e ac247399 9871ca90 7d4cff84 \
  1b8d1483 e38b847b 35643ceb 42ac49f3

# Deploys per day, whole project (Figure 2) — reproduces MINING_REPORT.md §3
export TZ=America/Los_Angeles
git -C /workspace/whomp-play log --pretty='%ad' --date=format:'%Y-%m-%d' | sort | uniq -c

git -C /workspace/whomp-play log --pretty='%s' | grep -cv '^play build '  # 1, not 0 (see mismatch above)
```
